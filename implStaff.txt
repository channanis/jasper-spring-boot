 @Operation(
            summary = "Get one contract by CIN and identifiers",
            description = "Retrieve a single contract by CIN, categorie, police, and matricule."
    )
    @ApiResponses(value = {
            @ApiResponse(responseCode = "200", description = "Contract found"),
            @ApiResponse(responseCode = "404", description = "Contract not found")
    })
    @GetMapping("/{cin}")
    public ContratMobileDTO getContract(
            @Parameter(description = "Customer CIN", required = true)
            @PathVariable String cin,
            @Parameter(description = "Contract category number", required = true)
            @RequestParam Integer categorie,
            @Parameter(description = "Contract adhesion number", required = true)
            @RequestParam String police,
            @Parameter(description = "Contract matricule")
            @RequestParam(required = false) Long matricule,
            @Parameter(description = "Data source: J or J-1", required = true)
            @RequestParam String source) throws CustomBadRequestException, NoResultException {
        log.info("Start resource getContract: cin={}, numCat={}, nadh={}, matricule={}", cin, categorie, police, matricule);
        ContratDetailService contratDetailService = contratDetailStrategyResolver.resolve(String.valueOf(categorie), SourceFlag.validate(source));
        ContratMobileDTO contract = contratDetailService.getcontratDetail(cin, categorie, police, matricule);
        log.info("End resource getContract: contract found for cin={}, numCat={}, nadh={}, matricule={}", cin, categorie, police, matricule);
        return contract;
    }
