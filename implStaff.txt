        log.info("Start service searchContracts: cin={}, type={}", cin, typeProduit);

        List<ContratMobile> contracts = new ArrayList<>();

        if (typeProduit == null || typeProduit.isBlank()) {
            contracts = contratRepository.findAllByCinIgnoreCase(cin);

        } else if (!ProductTypeEnum.EPARGNE.getLabel().equalsIgnoreCase(typeProduit)) {
            contracts = contratRepository.findAllByCinIgnoreCaseAndTypeProduitIn(cin, List.of(typeProduit));

        } else {
            log.info("Fetching EPARGNE contracts for CIN={} using offer configuration", cin);
            List<Offer> offersEpargnes = offerRepository.findByEpargne();
            log.info("Offers with EPARGNE flag found: {}", offersEpargnes.size());

            for (Offer offer : offersEpargnes) {
                contracts.addAll(contratRepository.findAllByCinIgnoreCaseAndNumCat(cin, offer.getNumCat()));
            }
        }

        List<AdherentStaff> staffContractsEntity = adherentStaffRepository.findByCinAndNumcatIgnoreCaseTrim(cin, ProduitConstants.DYNAMIQUE_ENTREPRISE);
        List<ContratMobile> staffContracts = adherentStaffMapper.toContratsMobileList(staffContractsEntity);

        Long maxId = getMaxId(contracts);
        setIdsForContractsWithNullIds(maxId, staffContracts);

        contracts.addAll(staffContracts);

        contracts = contracts.stream()
                .filter(c -> ProductTypeEnum.isAllowed(c.getTypeProduit()))
                .collect(Collectors.toCollection(ArrayList::new));

        var output = contratMobileMapper.toContractMobileList(contracts);
        log.info("End service searchContracts: total={}", output.size());
        return output;
    }

    default BigDecimal toBigDecimal(Double value) {
        return value != null ? BigDecimal.valueOf(value) : null;
    }

    default Double getMontantPremierVersement(AdherentStaff adherent) {
        return adherent.getCotisations().stream()
                .filter(c -> c.getDatecap() != null)
                .min(Comparator.comparing(CotisatiStaff::getDatecap))
                .map(CotisatiStaff::getCottotale)
                .orElse(null);
    }

    default LocalDate getLatestPmprclDate(AdherentStaff adherent) {
        return adherent.getPmprcls().stream()
                .map(PmprclStaff::getDatcalc)
                .max(LocalDate::compareTo)
                .orElse(null);
    }

    default Double getCapitalPartSalariale(AdherentStaff adherent) {
        return adherent.getPmprcls().stream()
                .max(Comparator.comparing(PmprclStaff::getDatcalc))
                .map(p -> p.getVprimps() != null ? p.getVprimps() : 0.0)
                .orElse(0.0);
    }

    default Double getCapitalPartPatronale(AdherentStaff adherent) {
        return adherent.getPmprcls().stream()
                .max(Comparator.comparing(PmprclStaff::getDatcalc))
                .map(p -> p.getVprimpp() != null ? p.getVprimpp() : 0.0)
                .orElse(0.0);
    }

    default Double getCapitalTotal(AdherentStaff adherent) {
        double partSalariale = getCapitalPartSalariale(adherent);
        double partPatronale = getCapitalPartPatronale(adherent);
        return BigDecimal.valueOf(partSalariale)
                .add(BigDecimal.valueOf(partPatronale))
                .setScale(2, RoundingMode.HALF_UP)
                .doubleValue();
    }

    default Double getAvanceNonRemboursee(AdherentStaff adherent) {
        return adherent.getPmprcls().stream()
                .max(Comparator.comparing(PmprclStaff::getDatcalc))
                .map(p -> p.getMtavanceTtc() != null ? p.getMtavanceTtc() : 0.0)
                .orElse(0.0);
    }

    @Named("mapFrequence")
    default String mapFrequence(String modp) {
        if (modp == null) return null;
        modp = modp.trim().toUpperCase();
        return switch (modp) {
            case "MENSUEL" -> "M";
            case "ANNUEL" -> "A";
            case "TRIMESTRIEL" -> "T";
            case "SEMESTRIEL" -> "S";
            default -> modp;
        };
    }

    default String formatCin(String cin) {
        if (cin == null) return null;
        return cin.toUpperCase().replace(" ", "").replace(".", "");
    }

    default String trim(String value) {
        return value != null ? value.trim() : null;
    }
