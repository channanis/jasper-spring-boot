@Override
    public PrintingDownloadOutputDTO downloadDocument(Long numCat, Long police, Long matricule, HttpServletRequest request) {

        log.info("Start printing doc with numContrat={}, numCat={}, matricule={}", new Object[]{police, numCat, matricule});

        // Vérification d'autorisation
        if (authorizationUtil.isCustomerNotAuthorized(
                request,
                police,
                numCat.intValue(),
                matricule)) {
            log.warn("Authorization failed for contrat={}, categorie={}, matricule={}", new Object[]{police, numCat, matricule});
            throw new ForbiddenException("Accès interdit", "FORBIDDEN");
        }

        VersementsResponse versements = businessContratClientMS.loadVersements(numCat, police, matricule);

        DocumentGenerationRequest generationRequest = new DocumentGenerationRequest();
        generationRequest.setFormat("PDF");

        switch (versements.getCategorieParente()) {
            case "BANCASS":
                BancassDocumentDTO bancass = VersementMapper.mapToBancass(versements);
                generationRequest.setCode("ATTESTATION_VERSEMENT_BANCASS");
                generationRequest.setData(bancass.toMap());
                break;

            case "GROUPE":
                GroupDocumentDTO group = VersementMapper.mapToGroup(versements);
                generationRequest.setCode("ATTESTATION_VERSEMENT_GROUPE");
                generationRequest.setData(group.toMap());
                break;

            case "INDIVIDUAL":
                IndividualDocumentDTO individual = VersementMapper.mapToIndividual(versements);
                generationRequest.setCode("ATTESTATION_VERSEMENT_INDIVIDUAL");
                generationRequest.setData(individual.toMap());
                break;

            default:
                throw new TechnicalException("Categorie non supportée");
        }

        PrintingDownloadOutputDTO generated = printingClientMS.generateDocument(generationRequest);
        log.info("End downloadContractVersement: fileName={}", generated.getFileName());
        return generated;
    }

        map.put("nom", nom);

    }
}

public class VersementMapper {

    public static BancassDocumentDTO mapToBancass(VersementsResponse v) {
        BancassDocumentDTO dto = new BancassDocumentDTO();
        dto.setCivilite(v.getCivilite());
        dto.setNom(v.getNom());
        dto.setPrenom(v.getPrenom());
        dto.setAdresse1(v.getAdresse1());
        dto.setAdresse2(v.getAdresse2());
        dto.setVille(v.getVille());
        dto.setNumeroAdhesion(v.getNumContrat() != null ? v.getNumContrat().toString() : null);
        dto.setDateEffet(v.getDateEffet() != null ? v.getDateEffet() : null);
        dto.setLibelleContrat(v.getLibCategore());
        dto.setNumeroContrat(v.getNumContrat() != null ? v.getNumContrat().toString() : null);
        dto.setEcheances(mapPayments(v.getVersements()));
        return dto;
    }

    public static GroupDocumentDTO mapToGroup(VersementsResponse v) {
        GroupDocumentDTO dto = new GroupDocumentDTO();
        dto.setCivilite(v.getCivilite());
        dto.setNom(v.getNom());
        dto.setAdresse1(v.getAdresse1());
        dto.setAdresse2(v.getAdresse2());
        dto.setVille(v.getVille());
        dto.setPays(v.getPays());
        dto.setNumeroAdhesion(v.getMatricule() != null ? v.getMatricule().toString() : null);
        dto.setLibelleContrat(v.getLibCategore());
        dto.setNumeroPolice(v.getPolice() != null ? v.getPolice().toString() : null);
        dto.setEcheances(mapPayments(v.getVersements()));
        return dto;
    }

    public static IndividualDocumentDTO mapToIndividual(VersementsResponse v) {
        IndividualDocumentDTO dto = new IndividualDocumentDTO();
        dto.setCivilite(v.getCivilite());
        dto.setNom(v.getNom());
        dto.setPrenom(v.getPrenom());
        dto.setAdresse1(v.getAdresse1());
        dto.setAdresse2(v.getAdresse2());
        dto.setVille(v.getVille());
        dto.setPays(v.getPays());
        dto.setNumeroAdhesion(v.getNumContrat() != null ? v.getNumContrat().toString() : null);
        dto.setLibelleContrat(v.getLibCategore());
        dto.setEcheances(mapPayments(v.getVersements()));
        return dto;
    }

    public static List<Map<String,Object>> mapPayments(List<Payment> payments) {
        List<Map<String,Object>> list = new ArrayList<>();
        if (payments != null) {
            for (Payment p : payments) {
                Map<String,Object> line = new HashMap<>();
                line.put("echeance", p.getEcheance() != null ? p.getEcheance() : null);
                line.put("montantVersement", p.getMontant());
                line.put("dateEncaissement", p.getDateEncaisement() != null ? p.getDateEncaisement().toString() : null);
                line.put("modePaiement", p.getModePaiement());
                list.add(line);
            }
        }
        return list;
    }

}
