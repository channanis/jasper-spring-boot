@Component
@Path("/private/prestation")
public class RestPrestation {
    private static final Logger log = LoggerFactory.getLogger(RestPrestation.class);
    @Autowired
    PrestationService prestationService;

    @POST
    @Path("/demandePrestation")
    @Produces(MediaType.APPLICATION_JSON)
    @Consumes(MediaType.MULTIPART_FORM_DATA)
    public Response add(@Context HttpServletRequest request, @MultipartForm PrestationFormBean prestationFormBean) throws IOException {
        FrontUserDetails user = (FrontUserDetails) request.getSession().getAttribute("user");
        if (user == null) {
            log.error("Error while  demandePrestation");
            return Utils.generateResponse();
        }
        try {
            log.debug("Start Adding prestation objet: " + prestationFormBean.getPrestation());
            log.debug(" Adding prestation file: " + Arrays.toString(prestationFormBean.getData()));
            prestationService.addPrestation(prestationFormBean,user.getUsername());
            Map<String, String> map = new HashMap<String, String>();
            map.put("code", "OK");
            log.debug("End Adding prestation");
            return Response.status(HttpServletResponse.SC_OK).entity(map).build();
        } catch (ServiceException e) {
            log.error("Error while adding prestation");
            return Response.status(HttpServletResponse.SC_OK).entity(e.toMap()).build();
        }

    }

}

@Override
    public void addPrestation(PrestationFormBean prestationFormBean, String login) throws IOException, ServiceException {
        ObjectMapper objectMapper = new ObjectMapper();
        PrestationBean prestation = objectMapper.readValue(prestationFormBean.getPrestation(), PrestationBean.class);
        if (prestationFormBean.getData() != null){
            FileValidationUtil.validateUploadedFile(prestationFormBean.getData(), Integer.parseInt(max_file_size), prestation.getFileExt());
        }
        Customer customer = customerService.findByUsername(login);
        String fileName = "";
        logger.debug(" start addPrestation service prestation <" + prestation.toString() + ">");

        /*********** Start test RG**********************/
        checkFieldDemandePrestation(customer, prestation);

        Contract contract = contractDao.getContract(customer.getCin(), prestation.getNumCat(), prestation.getPolice());

        checkContractNanti(contract);

        checkRGAvance(prestation, customer);

        checkRGRachatPartiel(prestation, contract);

        if (Constants.VIREMENT.equals(prestation.getModeReglement())) {
            logger.debug(" start test cas VIREMENT");

            if (org.apache.commons.lang.StringUtils.isBlank(prestation.getNumeroCompteBancaire())) {
                logger.error("Error while adding prestation FIELD_N_COMPTE_EMPTY");
                throw new ServiceException(ServiceExceptionCode.FIELD_N_COMPTE_EMPTY);
            }
            if (!prestation.getNumeroCompteBancaire().matches("[0-9]+")) {
                logger.error("Error while adding prestation FIELD_N_COMPTE_ONLY_DIGITS");
                throw new ServiceException(ServiceExceptionCode.FIELD_N_COMPTE_ONLY_DIGITS);
            }
            if (prestation.getNumeroCompteBancaire().length() != 24) {
                logger.error("Error while adding prestation FIELD_N_COMPTE_EXACT_24");
                throw new ServiceException(ServiceExceptionCode.FIELD_N_COMPTE_EXACT_24);
            }
            if (prestationFormBean.getData() == null) {
                logger.error("Error while adding prestation FIELD_FILE_EMPTY");
                throw new ServiceException(ServiceExceptionCode.FIELD_FILE_EMPTY);
            }
            if ((prestationFormBean.getData().length / 1000000) > Integer.parseInt(max_file_size)) {
                logger.error("Error while adding prestation FIELD_FILE_TOO_LARGE");
                throw new ServiceException(ServiceExceptionCode.FIELD_FILE_TOO_LARGE);
            }
            fileName = generatePrestationFileName(prestation);
            ftpFileUpload.upload(prestationFormBean.getData(), fileName, PRESTATION_FOLDER);
            logger.debug(" end test cas VIREMENT");
        }
        /************** END test RG ************/

        // create prestation
        createPrestation(prestation, customer, fileName, contract);

    }
private String generatePrestationFileName(PrestationBean prestation) {
        return "Prestation_" + Calendar.getInstance().getTime().getTime() + "." + prestation.getFileExt();
    }

    private float calculePercentage(float montant, Integer perc) {
        return (montant * perc) / 100;
    }

    private String generateUsr(Customer customer) {
        return (!org.apache.commons.lang.StringUtils.isBlank(customer.getNom()) && !org.apache.commons.lang.StringUtils.isBlank(customer.getPrenom())) ? customer.getPrenom().charAt(0) + "." + customer.getNom() : "";
    }

    private String generateTyp(String typeProcedure) {
        if (Constants.AVANCE.equals(typeProcedure)) {
            return "A";
        } else if (Constants.RACHAT_PARTIEL.equals(typeProcedure)) {
            return "RP";
        } else if (Constants.RACHAT_TOTAL.equals(typeProcedure)) {
            return "RT";
        }
        return "";
    }

    private void createPrestation(PrestationBean prestation, Customer customer, String fileName, Contract contract) {
        String uniqueID = UUID.randomUUID().toString();
        /* instancie la date*/
        Date date = new Date(System.currentTimeMillis());

        /*instancie la trace*/
        logger.debug("start instancie la trace");

        HistoryPrestation historyPrestation = new HistoryPrestation();
        historyPrestation.setDateModification(date);
        historyPrestation.setStatut(Constants.DEMANDE_MOBILE);
        historyPrestation.setModifierPar("User Mobile");
        logger.debug("start add prestation");
        Prestation prestationToSave = new Prestation(prestation.isVersementEntreprise(), prestation.getNomEntreprise(),
                prestation.isValeurMaximale(), prestation.getMantantOperation(), prestation.getModeReglement(),
                date, date, prestation.getNumCat(), prestation.getPolice(),
                Constants.DEMANDE_MOBILE, prestation.getTypeProcedure(), customer.getCin(), customer.getLogin(), uniqueID);
        if (!"".equals(fileName)) {
            prestationToSave.setReleveIdentiteBancaire(ma_marocainevie_selfcare_mobile_url_img_prestations + fileName);
        }

        prestationToSave.setUsr(generateUsr(customer));
        prestationToSave.setNumeroCompteBancaire(prestation.getNumeroCompteBancaire());
        prestationToSave.setTyp(generateTyp(prestation.getTypeProcedure()));
        prestationToSave.setToken(Utils.generateToken());
        prestationToSave.setLibNumCat(contract.getLIBNUMCAT());
        Prestation prestationSaved = prestationDao.addPrestation(prestationToSave);

        logger.debug("end add prestation");

        customer.setEmail(prestation.getEmail());
        customerDao.update(customer);
        /* ajout la trace */
        historyPrestation.setPrestation(prestationSaved);
        historyPrestationService.addHistoryPrestation(historyPrestation);
        logger.debug("end instancie la trace");
    }

    private void updatePrestation(PrestationRequest prestationRequest, Prestation prestation, String message, long time) {
        logger.debug("start updatePrestation");
        Date date = new Date(System.currentTimeMillis());
        prestation.setCodsit(prestationRequest.getStatut());
        prestation.setDateModification(date);
        prestation.setQuittance(prestationRequest.getQuittance());
        if (!"".equals(message)) {
            prestation.setCommentaire(Utils.createCommentPrestation(Constants.LMV, message, time));
        }
        prestationDao.update(prestation);

        logger.debug("end updatePrestation");
        /*instancie la trace*/
        logger.debug(" start instancie la trace");
        HistoryPrestation historyPrestation = new HistoryPrestation();
        historyPrestation.setDateModification(date);
        historyPrestation.setStatut(prestationRequest.getStatut());
        historyPrestation.setModifierPar(Constants.LMV);
        historyPrestationService.addHistoryPrestation(historyPrestation);
        logger.debug(" end instancie la trace");
    }

    private void checkFieldDemandePrestation(Customer customer, PrestationBean prestation) throws ServiceException {
        if (customer == null) {
            logger.error("Error while adding prestation CUSTOMER_EMPTY");
            throw new ServiceException(ServiceExceptionCode.CUSTOMER_EMPTY);
        }
        /* test if nom  prenom is empty */
        if (org.apache.commons.lang.StringUtils.isBlank(prestation.getNomPrenom())) {
            logger.error("Error while adding prestation FIELD_LAST_NAME_EMPTY");
            throw new ServiceException(ServiceExceptionCode.FIELD_LAST_NAME_EMPTY);
        }

        /* test if police is empty */
        if (org.apache.commons.lang.StringUtils.isBlank(prestation.getPolice())) {
            logger.error("Error while adding prestation FIELD_POLICE_EMPTY");
            throw new ServiceException(ServiceExceptionCode.FIELD_POLICE_EMPTY);
        }
        if (prestation.getNumCat() == null) {
            logger.error("Error while adding prestation NUMCAT_EMPTY");
            throw new ServiceException(ServiceExceptionCode.NUMCAT_EMPTY);
        }
        if (prestation.isVersementEntreprise() && org.apache.commons.lang.StringUtils.isBlank(prestation.getNomEntreprise())) {
            logger.error("Error while adding prestation FIELD_ENTREPRISE_EMPTY");
            throw new ServiceException(ServiceExceptionCode.FIELD_ENTREPRISE_EMPTY);
        }
        /* test if email is valide */
        if (!org.apache.commons.lang.StringUtils.isBlank(prestation.getEmail()) && !prestation.getEmail().matches(
                "[a-zA-Z0-9\\+\\.\\_\\%\\-\\+]{1,256}\\@[a-zA-Z0-9][a-zA-Z0-9\\-]{0,64}(\\.[a-zA-Z0-9][a-zA-Z0-9\\-]{0,25})+"
        )) {
            logger.error("Error while adding prestation FIELD_EMAIL_FORMAT_ERROR");
            throw new ServiceException(ServiceExceptionCode.FIELD_EMAIL_FORMAT_ERROR);
        }

        /* test if date  birthady is empty */
        if (customer.getDateNaissance() == null) {
            logger.error("Error while adding prestation FIELD_BIRTHDATE_EMPTY");
            throw new ServiceException(ServiceExceptionCode.FIELD_BIRTHDATE_EMPTY);
        }
        /* test if date souscription is empty */
        if (prestation.getDateSous() == null) {
            logger.error("Error while adding prestation FIELD_DATE_SOUSCRIPTION_EMPTY");
            throw new ServiceException(ServiceExceptionCode.FIELD_DATE_SOUSCRIPTION_EMPTY);
        }
    }

    private void checkFieldMajPrestation(Prestation prestation, PrestationRequest prestationRequest, String token) throws ServiceException, AuthenticationExeption {
        if (prestation == null) {
            logger.error("majPrestation error PRESTATION_EMPTY");
            throw new ServiceException(ServiceExceptionCode.PRESTATION_EMPTY);
        }
        if (!token.equals(prestation.getToken())) {
            throw new AuthenticationExeption(AuthenticationExceptionCode.FORBIDDEN);
        }
        logger.debug("prestation from database mobile" + prestation.toString());
        if (prestationRequest.getStatut() != null && prestationRequest.getStatut().equals(prestation.getCodsit())) {
            logger.error("majPrestation error STATUS_EXIST");
            throw new ServiceException(ServiceExceptionCode.STATUS_EXIST);
        }
    }

    private void checkContractNanti(Contract contract) throws ServiceException {
        logger.debug(" start test contract ");

        if (contract == null) {
            logger.error("Error while adding prestation CONTRACT_EMPTY");
            throw new ServiceException(ServiceExceptionCode.CONTRACT_EMPTY);
        }
        logger.debug(" contract <" + contract.toString() + ">");

        if (contract.getDefntsm() != null && contract.getNtsmt()) {
            logger.error("Error while adding prestation CONTRACT_NANTI");
            throw new ServiceException(ServiceExceptionCode.CONTRACT_NANTI);
        }
        logger.debug(" end test contract ");
    }

    private void checkRGAvance(PrestationBean prestation, Customer customer) throws ServiceException {
        /* calcule age et anciente */
        if (Constants.AVANCE.equals(prestation.getTypeProcedure())) {
            logger.debug("start test cas avance");
            logger.debug("arg prestation <" + prestation.toString() + ">");
            logger.debug("arg customer <" + customer.toString() + ">");

            int age = Utils.getYears(customer.getDateNaissance());
            int dureContart = Utils.getYears(prestation.getDateSous());
            logger.debug("age <" + age + ">");
            logger.debug("dureContart <" + dureContart + ">");
            if (age < 50) {
                logger.error("Error while adding prestation AGE_CRITERIA");
                throw new ServiceException(ServiceExceptionCode.AGE_CRITERIA);
            }
            if (dureContart < 8) {
                logger.error("Error while adding prestation DURE_CONTRAT_CRITERIA");
                throw new ServiceException(ServiceExceptionCode.DURE_CONTRAT_CRITERIA);
            }
            //RG calcul 80%
            float montant = calculePercentage(prestation.getMtencour(), 80);
            logger.debug("montant <" + montant + ">");
            if (!prestation.isValeurMaximale()) {
                if (prestation.getMantantOperation() > montant) {
                    logger.error("Error while adding prestation MONTANT_SUPERIEUR_80_ERROR");
                    throw new ServiceException(ServiceExceptionCode.MONTANT_SUPERIEUR_80_ERROR);
                }
            } else {
                prestation.setMantantOperation(montant);
            }
            //RG avance non rembourssé
            if (prestation.getMtavce() > 0) {
                logger.error("Error while adding prestation AVANCE_NON_REMBOURSE_ERROR");
                throw new ServiceException(ServiceExceptionCode.AVANCE_NON_REMBOURSE_ERROR);
            }
            logger.debug("end test cas avance");
        }
    }

    private void checkRGRachatPartiel(PrestationBean prestation, Contract contract) throws ServiceException {

        if (Constants.RACHAT_PARTIEL.equals(prestation.getTypeProcedure())) {
            logger.debug("start test cas RACHAT_PARTIEL arg prestation <" + prestation.toString() + ">");
            logger.debug("start test cas RACHAT_PARTIEL arg  contart <" + contract.toString() + ">");
            //RG calcul 50%
            float montant = calculePercentage(prestation.getMtencour(), 50);
            logger.debug("montant <" + montant + ">");
            if (!prestation.isValeurMaximale()) {
                if (prestation.getMantantOperation() > montant) {
                    logger.error("Error while adding prestation MONTANT_SUPERIEUR_50_ERROR");
                    throw new ServiceException(ServiceExceptionCode.MONTANT_SUPERIEUR_50_ERROR);
                }
            } else {
                prestation.setMantantOperation(montant);
            }

            //RG avance encours
            if (prestation.getMtavce() > 0) {
                logger.error("Error while adding prestation AVANCE_ENCOURS_ERROR");
                throw new ServiceException(ServiceExceptionCode.AVANCE_ENCOURS_ERROR);
            }

            // RG nombre de rachat partiel
            if (contract.getNbrachat() != null && contract.getNbrachat() > 3) {
                logger.error("Error while adding prestation NOMBRE_RACHAT_ERROR");
                throw new ServiceException(ServiceExceptionCode.NOMBRE_RACHAT_ERROR);
            }
            logger.debug("end test cas RACHAT_PARTIEL");
        }
    }

@Value("${ma_marocainevie_selfcare_mobile_directoryBase}")
    private String ma_marocainevie_selfcare_mobile_directoryBase;

    @Value("${ma_marocainevie_selfcare_mobile_login}")
    private String ma_marocainevie_selfcare_mobile_login;

    @Value("${ma_marocainevie_selfcare_mobile_password}")
    private String ma_marocainevie_selfcare_mobile_password;

    @Value("${ma_marocainevie_selfcare_mobile_cert}")
    private String ma_marocainevie_selfcare_mobile_cert;

    @Value("${ma_marocainevie_selfcare_mobile_server}")
    private String ma_marocainevie_selfcare_mobile_server;

    @Value("${ma_marocainevie_selfcare_mobile_port}")
    private String ma_marocainevie_selfcare_mobile_port;

    @Value("${ma_marocainevie_selfcare_mobile_timeout}")
    private String ma_marocainevie_selfcare_mobile_timeout;

public boolean upload(byte[] image, String fileName, String dossier) throws ServiceException {
        log.info("FTP upload initiated : fileName = {}, target folder = {}", fileName, dossier);
        String ftpBaseDirectory = ma_marocainevie_selfcare_mobile_directoryBase + dossier;
        String ftpLogin = ma_marocainevie_selfcare_mobile_login;
        String ftpPassword = ma_marocainevie_selfcare_mobile_password;
        String ftpServer = ma_marocainevie_selfcare_mobile_server;
        String ftpPort = ma_marocainevie_selfcare_mobile_port;
        ftpPort = ftpPort != null && !ftpPort.isEmpty() ? ftpPort : "21";
        String ftpTimeout = ma_marocainevie_selfcare_mobile_timeout;
        String ftpCertFile = ma_marocainevie_selfcare_mobile_cert;

        if (ftpCertFile != null && ftpCertFile.isEmpty()) {
            ftpCertFile = null;
            log.info("FTP cert file was empty, reset to null");
        }

        if (ftpPassword != null && ftpPassword.isEmpty()) {
            ftpPassword = null;
            log.info("FTP password was empty, reset to null");
        }
        log.info("Delegating to internal upload handler");
        return upload(image, fileName, ftpBaseDirectory, ftpLogin, ftpPassword, ftpCertFile, ftpServer, ftpPort, ftpTimeout);
    }
