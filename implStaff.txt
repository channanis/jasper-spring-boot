@Operation(
            summary = "Search contracts by CIN and type produit",
            description = "Retrieve contracts by CIN. filter by typeProduit."
    )
    @ApiResponses(value = {
            @ApiResponse(responseCode = "200", description = "Contracts found"),
            @ApiResponse(responseCode = "400", description = "Missing/invalid X-Canal header or source param"),
            @ApiResponse(responseCode = "404", description = "No contracts found")
    })
    @GetMapping("/{cin}/search")
    public List<ContratMobileDTO> searchContracts(
            @Parameter(description = "Customer CIN", required = true)
            @PathVariable String cin,
            @Parameter(description = "Optional contract type")
            @RequestParam(required = false) String typeProduit,
            @Parameter(description = "Data source: J or J-1", required = true)
            @RequestParam String source,
            @RequestHeader(CanalEnum.HEADER_NAME) String canalHeader) throws CustomBadRequestException {

        CanalEnum canal = contratServiceFactory.resolveCanal(canalHeader);

        log.info("Start resource searchContracts: cin={}, type={}, canal={}, source={}",
                cin, typeProduit, canal, source);

        ContractService contractService = contratServiceFactory.resolve(SourceFlag.validate(source));
        List<ContratMobileDTO> contracts = contractService.searchByCinAndTypeProduit(cin, typeProduit, canal);

        log.info("End resource searchContracts: cin={}, type={}, total={}", cin, typeProduit, contracts.size());
        return contracts;
    }
@Component
public class ContratServiceFactory {

    private final Map<SourceFlag, ContractService> strategies;

    public ContratServiceFactory(List<ContractService> services) {
        this.strategies = services.stream()
                .collect(Collectors.toMap(ContractService::getSource, Function.identity()));
    }

    public ContractService resolve(SourceFlag source) {
        ContractService service = strategies.get(source);
        if (service == null) {
            throw new IllegalStateException("Aucune implémentation ContratService pour la source: " + source);
        }
        return service;
    }

    public CanalEnum resolveCanal(String canalHeader) {
        try {
            return CanalEnum.valueOf(canalHeader.trim().toUpperCase());
        } catch (IllegalArgumentException | NullPointerException e) {
            throw new InvalidCanalException(canalHeader);
        }
    }
}

public interface ContractService {


    List<ContratMobileDTO> searchByCin(String cin, CanalEnum canal);
    List<ContratMobileDTO> searchByCinAndTypeProduit(String cin, String typeProduit, CanalEnum canal);

    /**
     * Source gérée par cette implémentation — utilisée par {@link ContratServiceFactory}
     * pour sélectionner la bonne stratégie.
     */
    SourceFlag getSource();
}
@Service
@Slf4j
public class ContractServiceJMoins1Impl implements ContractService {
     private final CanalStrategyResolver canalStrategyResolver;

    public ContractServiceJMoins1Impl(BffLmvCanalStrategyJMoins1 bffLmvCanalStrategy,
                                       BffMobiOneCanalStrategyJMoins1 bffMobiOneCanalStrategy) {
         this.canalStrategyResolver = new CanalStrategyResolver(List.of(bffLmvCanalStrategy, bffMobiOneCanalStrategy));
    }

    @Override
    public List<ContratMobileDTO> searchByCin(String cin, CanalEnum canal) {
        return this.canalStrategyResolver.resolve(canal).searchByCin(cin);
    }

    @Override
    public List<ContratMobileDTO> searchByCinAndTypeProduit(String cin, String typeProduit, CanalEnum canal) {
        return this.canalStrategyResolver.resolve(canal).searchByCinAndTypeProduit(cin,typeProduit);
    }

    @Override
    public SourceFlag getSource() {
        return SourceFlag.J_MINUS_1;
    }

}

public interface CanalSearchStrategy {
    List<ContratMobileDTO> searchByCin(String cin);
    List<ContratMobileDTO> searchByCinAndTypeProduit(String cin, String typeProduit);
    CanalEnum getCnale();
}

@Component
@Slf4j
@RequiredArgsConstructor
public class BffLmvCanalStrategyJMoins1 implements CanalSearchStrategy {
    private final ContratMobileMapper contratMobileMapper;
    private final ContratStaffEnricher contratStaffEnricher;
    private final ContratMobileRepository contratRepository;

    @Override
    @Transactional(readOnly = true)
    public List<ContratMobileDTO> searchByCin(String cin) {
        log.info("Start service searchContracts mobile by cin={}", cin);

        List<ContratMobile> contracts = contratRepository.findAllByCinIgnoreCase(cin);

        contracts.addAll(contratStaffEnricher.fetchStaffContracts(cin, contracts));

        List<ContratMobileDTO> output = contratMobileMapper.toContractMobileList(contratStaffEnricher.filterAllowed(contracts));
        log.info("End service searchContracts mobile by cin: total={}", output.size());
        return output;
    }

    @Override
    @Transactional(readOnly = true)
    public List<ContratMobileDTO> searchByCinAndTypeProduit(String cin, String typeProduit) {
        log.info("Start service searchContracts mobile: cin={}, type={}", cin, typeProduit);

        List<ContratMobile> contracts = fetchContractsMobileByType(cin, typeProduit);

        contracts.addAll(contratStaffEnricher.fetchStaffContracts(cin, contracts));

        List<ContratMobileDTO> output = contratMobileMapper.toContractMobileList(contratStaffEnricher.filterAllowed(contracts));
        log.info("End service searchContracts mobile: total={}", output.size());
        return output;
    }

    @Override
    public CanalEnum getCnale() {
        return CanalEnum.BFF_LMV;
    }

    private List<ContratMobile> fetchContractsMobileByType(String cin, String typeProduit) {
        if (!ProductTypeEnum.EPARGNE.getLabel().equalsIgnoreCase(typeProduit)) {
            return new ArrayList<>(contratRepository.findAllByCinIgnoreCaseAndTypeProduitIn(cin, List.of(typeProduit)));
        }
        return new ArrayList<>(contratRepository.findSavingContracts(cin));
    }
}
